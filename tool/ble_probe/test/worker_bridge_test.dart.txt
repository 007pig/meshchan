Future<WorkerFixture> _fixture({bool syntheticFaults = false}) async {
  final directory = await createCooperationStorageFixture();
  final fixture = WorkerFixture(directory, syntheticFaults: syntheticFaults);
  addTearDown(() async {
    final outcome = await fixture.close();
    if (outcome.workerExited) await directory.delete(recursive: true);
    expect(outcome.canExit, isTrue);
  });
  await fixture.start();
  return fixture;
}

Iterable<WorkerBridgeFact> _facts(
  WorkerFixture fixture,
  WorkerBridgeFactKind kind, {
  int? generation,
}) => fixture.facts.where(
  (fact) =>
      fact.kind == kind &&
      (generation == null || fact.nativeGeneration == generation),
);

Future<void> _settled(WorkerFixture fixture, DeviceCheckHandle handle) =>
    fixture.until(() {
      final snapshot = fixture.runtime.snapshot.deviceCheck;
      return snapshot?.id == handle.id &&
          snapshot?.settled == true &&
          fixture.runtime.snapshot.gate != DeviceGateState.occupied;
    });

enum _DelayedDelivery { finalFrame, finalWriteResult, close }

void main() {
  test(
    'spawned session handles both write/response orders and a fresh lease',
    () async {
      final fixture = await _fixture(syntheticFaults: true);
      final runtime = fixture.runtime;
      await writeCooperationStorageFixture(runtime);
      final first = HeldRootTransport(1);
      final check = fixture.begin(first);
      await check.accepted;
      await first.opened.future.timeout(fixtureWait);
      expect(first.openCalls, 1);
      expect(first.writes, isEmpty);
      expect(runtime.snapshot.gate, DeviceGateState.occupied);
      await expectLater(
        runtime.request(
          const StartDeviceCheck(99, 'unmapped', DeviceCheckOptions()),
        ),
        throwsA(isA<RuntimeFault>()),
      );
      expect(first.openCalls, 1);

      first.openResult.complete();
      await fixture.until(() => first.writes.length == 1);
      expect(first.writes.single, [0x16, meshCoreSessionAppTargetVersion]);
      first.emit(selfInfo: false, mutateAfterSend: true);
      await fixture.until(
        () => _facts(fixture, WorkerBridgeFactKind.frameWorker).isNotEmpty,
      );
      expect(first.writes.length, 1);
      expect(runtime.snapshot.deviceCheck?.info, isNull);
      first.writeResults.single.complete();
      await fixture.until(() => first.writes.length == 2);
      expect(first.writes.last, encodeAppStart(appName: 'Meshchan'));
      expect(first.writes.last.length, 16);
      first.writeResults.last.complete();
      await fixture.until(
        () =>
            _facts(fixture, WorkerBridgeFactKind.write)
                .where((fact) => fact.status == WorkerBridgeStatus.confirmed)
                .length >=
            2,
      );
      expect(runtime.snapshot.deviceCheck?.info, isNull);
      first.emit(selfInfo: true, mutateAfterSend: true);
      await first.closing.future.timeout(fixtureWait);
      await fixture.until(() => runtime.snapshot.deviceCheck?.info != null);
      expect(runtime.snapshot.deviceCheck?.info?.nodeName, privateNodeName);
      expect(runtime.snapshot.deviceCheck?.settled, isFalse);
      expect(runtime.snapshot.gate, DeviceGateState.occupied);
      first.closeResult.complete();
      await _settled(fixture, check);
      expect(runtime.snapshot.deviceCheck?.released, isTrue);
      expect(runtime.snapshot.gate, DeviceGateState.free);
      expect(first.closeCalls, 1);

      final workerFrames = _facts(
        fixture,
        WorkerBridgeFactKind.frameWorker,
        generation: 1,
      ).toList();
      expect(workerFrames.map((fact) => fact.nativeSequence), [2, 4]);
      expect(workerFrames.map((fact) => fact.opcode), [0x0d, 0x05]);
      expect(workerFrames.map((fact) => fact.length), [82, 64]);
      expect(workerFrames.map((fact) => fact.nativeLength), [82, 64]);
      expect(workerFrames.map((fact) => fact.expectedLength), [82, 64]);

      final second = HeldRootTransport(2)..openResult.complete();
      fixture.bridge.injectFixtureWireFault(WorkerBridgeWireFault.staleRequest);
      final followOn = fixture.begin(second);
      expect(followOn.id, isNot(check.id));
      await followOn.accepted;
      await expectLater(
        runtime.cancelDeviceCheck(check.id),
        throwsA(isA<RuntimeFault>()),
      );
      for (var index = 0; index < 2; index++) {
        await fixture.until(() => second.writes.length > index);
        if (index == 0) {
          fixture.bridge.injectFixtureWireFault(
            WorkerBridgeWireFault.staleFrame,
          );
        }
        second.writeResults[index].complete();
        second.emit(selfInfo: index == 1);
      }
      await second.closing.future.timeout(fixtureWait);
      second.closeResult.complete();
      await _settled(fixture, followOn);
      expect(runtime.snapshot.deviceCheck?.id, followOn.id);
      expect(runtime.snapshot.deviceCheck?.info?.nodeName, privateNodeName);
      expect(runtime.snapshot.deviceCheck?.released, isTrue);
      expect(runtime.snapshot.gate, DeviceGateState.free);
      expect(second.openCalls, 1);
      expect(second.closeCalls, 1);
      expect(_facts(fixture, WorkerBridgeFactKind.stale), isNotEmpty);

      final outcome = await fixture.close();
      expect(outcome.resourcesReleased, isTrue);
      expect(outcome.workerExited, isTrue);
      expect(runtime.snapshot.owner, OwnerFact.released);
      expect(runtime.pendingCount, 0);
      expect(runtime.openPortCount, 0);
      expect(_facts(fixture, WorkerBridgeFactKind.releaseReport), isNotEmpty);
      expect(
        _facts(fixture, WorkerBridgeFactKind.releaseAcknowledged),
        isNotEmpty,
      );
      expect(
        _facts(fixture, WorkerBridgeFactKind.workerPortsClosed),
        isNotEmpty,
      );
      expect(_facts(fixture, WorkerBridgeFactKind.workerExit), isNotEmpty);
      expect(_facts(fixture, WorkerBridgeFactKind.rootPortClosed), isNotEmpty);
    },
  );

  test(
    'matching response before a failed write cannot pass or dispatch APP_START',
    () async {
      late final WorkerFixture fixture;
      final transport = HeldRootTransport(1)..openResult.complete();
      final console = <String>[];
      await runZoned(
        () async {
          fixture = await _fixture();
          final check = fixture.begin(transport);
          await check.accepted;
          await fixture.until(() => transport.writes.length == 1);
          transport.emit(selfInfo: false);
          await fixture.until(
            () => _facts(fixture, WorkerBridgeFactKind.frameWorker).isNotEmpty,
          );
          transport.writeResults.single.completeError(
            StateError(privateMarker),
          );
          await transport.closing.future.timeout(fixtureWait);
          transport.closeResult.complete();
          await _settled(fixture, check);
          final snapshot = fixture.runtime.snapshot.deviceCheck!;
          expect(snapshot.info, isNull);
          expect(snapshot.failure?.kind, 'transport');
          expect(snapshot.failure?.commandOpcode, 0x16);
          expect(snapshot.released, isTrue);
          expect(transport.writes, hasLength(1));
          expect(
            _facts(
              fixture,
              WorkerBridgeFactKind.write,
            ).any((fact) => fact.status == WorkerBridgeStatus.failed),
            isTrue,
          );
        },
        zoneSpecification: ZoneSpecification(
          print: (_, _, _, value) => console.add(value),
        ),
      );
      final structural = jsonEncode([
        for (final fact in fixture.facts)
          {'kind': fact.kind.name, 'category': fact.category?.toString()},
        ...console,
      ]);
      expect(structural, isNot(contains(privateMarker)));
    },
  );

  test('Cancel requests close while Open is pending', () async {
    final fixture = await _fixture();
    final transport = HeldRootTransport(1);
    final check = fixture.begin(transport);
    await check.accepted;
    await transport.opened.future.timeout(fixtureWait);
    await fixture.runtime.cancelDeviceCheck(check.id);
    await transport.closing.future.timeout(fixtureWait);
    expect(transport.openResult.isCompleted, isFalse);
    expect(transport.closeCalls, 1);
    expect(transport.writes, isEmpty);
    transport.unblock();
    await _settled(fixture, check);
    final snapshot = fixture.runtime.snapshot.deviceCheck!;
    expect(snapshot.cancelApplied, isTrue);
    expect(snapshot.info, isNull);
    expect(snapshot.released, isTrue);
    expect(fixture.runtime.snapshot.gate, DeviceGateState.free);
    expect(transport.writes, isEmpty);
  });

  test('Cancel does not wait for a pending write', () async {
    final fixture = await _fixture();
    final transport = HeldRootTransport(1)..openResult.complete();
    final check = fixture.begin(transport);
    await check.accepted;
    await fixture.until(() => transport.writes.length == 1);
    final write = transport.writeResults.single;
    await fixture.runtime.cancelDeviceCheck(check.id);
    await transport.closing.future.timeout(fixtureWait);
    expect(write.isCompleted, isFalse);
    transport.unblock();
    await _settled(fixture, check);
    final snapshot = fixture.runtime.snapshot.deviceCheck!;
    expect(snapshot.cancelApplied, isTrue);
    expect(snapshot.info, isNull);
    expect(snapshot.released, isTrue);
    expect(transport.writes, hasLength(1));
    expect(transport.closeCalls, 1);
  });

  test(
    'expired preparation budget does not reopen at worker dispatch',
    () async {
      final fixture = await _fixture();
      final transport = HeldRootTransport(1)..closeResult.complete();
      final token = fixture.bridge.prepareAttempt(
        nativeGeneration: transport.generation,
        transport: transport,
        receiptOf: (_) => null,
        provenanceConfirmed: () => true,
        initializationBudget: const Duration(milliseconds: 20),
      );
      await Future<void>.delayed(const Duration(milliseconds: 40));
      final check = fixture.runtime.startDeviceCheck(
        token,
        const DeviceCheckOptions(),
      );
      await check.accepted;
      await _settled(fixture, check);
      expect(transport.openCalls, 0);
      expect(transport.writes, isEmpty);
      expect(transport.closeCalls, 1);
      expect(fixture.runtime.snapshot.deviceCheck?.released, isTrue);
      expect(
        _facts(
          fixture,
          WorkerBridgeFactKind.transportFault,
        ).map((fact) => fact.category),
        contains(WorkerBridgeFailure.initializationDeadline),
      );
    },
  );

  for (final (delayed, deadline) in [
    for (final delayed in _DelayedDelivery.values)
      (delayed, WorkerBridgeFailure.initializationDeadline),
    (_DelayedDelivery.finalFrame, WorkerBridgeFailure.requestDeadline),
    (_DelayedDelivery.finalWriteResult, WorkerBridgeFailure.requestDeadline),
    (_DelayedDelivery.close, WorkerBridgeFailure.requestDeadline),
  ]) {
    final delayClose = delayed == _DelayedDelivery.close;
    final requestDeadline = deadline == WorkerBridgeFailure.requestDeadline;
    final budgetName = requestDeadline ? 'request' : 'initialization';
    test(
      switch (delayed) {
        _DelayedDelivery.finalFrame =>
          'worker rejects a final response queued past the $budgetName deadline',
        _DelayedDelivery.finalWriteResult =>
          'worker rejects a final write result queued past the $budgetName deadline',
        _DelayedDelivery.close =>
          'on-time worker readiness survives Close past the $budgetName deadline',
      },
      () async {
        final fixture = await _fixture(syntheticFaults: true);
        final bridge = fixture.bridge;
        final transport = HeldRootTransport(1)..openResult.complete();
        const budget = Duration(seconds: 1);
        final check = requestDeadline
            ? fixture.begin(transport, requestBudget: budget)
            : fixture.begin(transport, initializationBudget: budget);
        await check.accepted;
        await fixture.until(() => transport.writes.length == 1);
        transport.writeResults.single.complete();
        transport.emit(selfInfo: false);
        await fixture.until(() => transport.writes.length == 2);
        final resume = Completer<void>();
        try {
          if (delayed == _DelayedDelivery.finalWriteResult) {
            transport.emit(selfInfo: true);
            await fixture.until(
              () =>
                  _facts(fixture, WorkerBridgeFactKind.frameWorker).length == 2,
            );
            bridge.pauseFixtureWorker(budget);
            transport.writeResults.last.complete();
          } else {
            transport.writeResults.last.complete();
            await fixture.until(
              () =>
                  _facts(fixture, WorkerBridgeFactKind.write)
                      .where(
                        (fact) => fact.status == WorkerBridgeStatus.confirmed,
                      )
                      .length ==
                  2,
            );
            if (delayClose) {
              bridge.pauseFixtureReceipts(resume.future);
            } else {
              // This control and the frame share the worker SendPort.
              bridge.pauseFixtureWorker(budget);
            }
            transport.emit(selfInfo: true);
          }
          expect(_facts(fixture, WorkerBridgeFactKind.frameRoot), hasLength(2));
          if (delayClose) {
            await fixture.until(
              () => fixture.runtime.snapshot.deviceCheck?.info != null,
            );
          }
          if (delayClose && requestDeadline) {
            // A completed native exchange has no outstanding request timer.
            await Future<void>.delayed(budget);
            expect(transport.closing.isCompleted, isFalse);
          } else {
            await transport.closing.future.timeout(fixtureWait);
          }
          transport.closeResult.complete();
        } finally {
          resume.complete();
        }
        await _settled(fixture, check);

        final snapshot = fixture.runtime.snapshot.deviceCheck!;
        expect(snapshot.info != null, delayClose);
        expect(bridge.initializationOnTime, delayClose);
        expect(snapshot.released, isTrue);
        expect(bridge.release, WorkerBridgeRelease.confirmed);
        expect(transport.closeCalls, 1);
        final deadlineFaults = _facts(
          fixture,
          WorkerBridgeFactKind.transportFault,
        ).where((fact) => fact.category == deadline);
        expect(deadlineFaults, hasLength(delayClose ? 0 : 1));
      },
    );
  }

  for (final missingResponse in [false, true]) {
    final stalledPart = missingResponse ? 'response' : 'write';
    test(
      'request watchdog closes a stalled $stalledPart while the worker is paused',
      () async {
        final fixture = await _fixture(syntheticFaults: true);
        final bridge = fixture.bridge;
        final transport = HeldRootTransport(1)..openResult.complete();
        final check = fixture.begin(
          transport,
          requestBudget: const Duration(seconds: 1),
        );
        await check.accepted;
        await fixture.until(() => transport.writes.length == 1);
        if (missingResponse) {
          transport.writeResults.single.complete();
          await fixture.until(
            () => _facts(
              fixture,
              WorkerBridgeFactKind.write,
            ).any((fact) => fact.status == WorkerBridgeStatus.confirmed),
          );
        } else {
          transport.emit(selfInfo: false);
          await fixture.until(
            () => _facts(fixture, WorkerBridgeFactKind.frameWorker).isNotEmpty,
          );
        }

        bridge.pauseFixtureWorker(const Duration(seconds: 2));
        await transport.closing.future.timeout(
          const Duration(milliseconds: 1500),
        );
        expect(transport.writeResults.single.isCompleted, missingResponse);
        expect(
          _facts(
            fixture,
            WorkerBridgeFactKind.close,
          ).where((fact) => fact.status == WorkerBridgeStatus.requested),
          isEmpty,
        );
        expect(
          _facts(fixture, WorkerBridgeFactKind.transportFault).single.category,
          WorkerBridgeFailure.requestDeadline,
        );
        transport.unblock();
        await _settled(fixture, check);
        expect(fixture.runtime.snapshot.deviceCheck?.released, isTrue);
        expect(transport.closeCalls, 1);
      },
    );
  }

  test(
    'missing provenance prevents either full frame from succeeding',
    () async {
      final fixture = await _fixture();
      final transport = HeldRootTransport(1)..openResult.complete();
      final check = fixture.begin(transport);
      await check.accepted;
      await fixture.until(() => transport.writes.length == 1);
      transport.writeResults.single.complete();
      transport.emit(selfInfo: false, withReceipt: false);
      await transport.closing.future.timeout(fixtureWait);
      transport.closeResult.complete();
      await _settled(fixture, check);
      expect(fixture.runtime.snapshot.deviceCheck?.info, isNull);
      expect(fixture.runtime.snapshot.deviceCheck?.released, isTrue);
      expect(_facts(fixture, WorkerBridgeFactKind.frameWorker), isEmpty);
      expect(transport.writes, hasLength(1));
    },
  );

  test(
    'late failed provenance preserves logical result and actual release',
    () async {
      final fixture = await _fixture();
      final transport = HeldRootTransport(1)..openResult.complete();
      final check = fixture.begin(transport);
      await check.accepted;
      for (var index = 0; index < 2; index++) {
        await fixture.until(() => transport.writes.length > index);
        transport.emit(selfInfo: index == 1);
        transport.writeResults[index].complete();
      }
      await transport.closing.future.timeout(fixtureWait);
      await fixture.until(
        () => fixture.runtime.snapshot.deviceCheck?.info != null,
      );
      transport.provenance = false;
      transport.closeResult.complete();
      await _settled(fixture, check);
      expect(
        fixture.runtime.snapshot.deviceCheck?.info?.nodeName,
        privateNodeName,
      );
      expect(fixture.runtime.snapshot.deviceCheck?.released, isTrue);
      expect(fixture.runtime.snapshot.gate, DeviceGateState.free);
      expect(fixture.bridge.provenanceConfirmed, isFalse);
      expect(fixture.bridge.canPrepareAttempt, isFalse);
      expect(transport.closeCalls, 1);
    },
  );
  for (final fault in [
    WorkerBridgeWireFault.duplicateRequest,
    WorkerBridgeWireFault.missingFrame,
    WorkerBridgeWireFault.duplicateFrame,
    WorkerBridgeWireFault.reorderedFrame,
  ]) {
    test(
      '${fault.name} fails without a second write or implicit retry',
      () async {
        final fixture = await _fixture(syntheticFaults: true);
        final transport = HeldRootTransport(1)..openResult.complete();
        if (fault == WorkerBridgeWireFault.duplicateRequest) {
          fixture.bridge.injectFixtureWireFault(fault);
        }
        final check = fixture.begin(transport);
        await check.accepted;
        if (fault != WorkerBridgeWireFault.duplicateRequest) {
          await fixture.until(() => transport.writes.length == 1);
          fixture.bridge.injectFixtureWireFault(fault);
          transport.emit(selfInfo: false);
          // A higher WriteResult sequence reveals a gap even without a second
          // notification.
          if (fault == WorkerBridgeWireFault.missingFrame) {
            transport.writeResults.single.complete();
          }
        }
        await transport.closing.future.timeout(fixtureWait);
        transport.unblock();
        await _settled(fixture, check);
        final snapshot = fixture.runtime.snapshot.deviceCheck!;
        expect(snapshot.info, isNull);
        expect(snapshot.failure, isNotNull);
        expect(snapshot.released, isTrue);
        expect(fixture.bridge.canPrepareAttempt, isFalse);
        expect(transport.openCalls, 1);
        expect(transport.closeCalls, 1);
        expect(transport.writes.length, lessThanOrEqualTo(1));
      },
    );
  }
}
